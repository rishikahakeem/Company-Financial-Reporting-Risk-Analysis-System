import joblib
import pandas as pd
import numpy as np
import os

class MLRiskPredictor:
    def __init__(self, model_type='neural_network'):
        # Get the path to the models folder
        base_dir = os.path.dirname(os.path.abspath(__file__))
        models_dir = os.path.join(base_dir, 'models')
        
        # Load models based on type
        if model_type == 'logistic_regression':
            self.model = joblib.load(os.path.join(models_dir, 'logistic_regression_model.pkl'))
        elif model_type == 'random_forest':
            self.model = joblib.load(os.path.join(models_dir, 'random_forest_model.pkl'))
        else:
            self.model = joblib.load(os.path.join(models_dir, 'neural_network_model.pkl'))
        
        # Load preprocessing objects
        self.scaler = joblib.load(os.path.join(models_dir, 'scaler.pkl'))
        self.imputer = joblib.load(os.path.join(models_dir, 'imputer.pkl'))
        
        # Load feature names
        with open(os.path.join(models_dir, 'features.txt'), 'r') as f:
            self.features = [line.strip() for line in f.readlines()]
        
        print(f"✅ Model loaded: {model_type}")
        print(f"📊 Using {len(self.features)} features")
    
    def prepare_features(self, df):
        """Extract features from uploaded data"""
        feature_df = pd.DataFrame()
        for feature in self.features:
            if feature in df.columns:
                feature_df[feature] = df[feature]
            else:
                feature_df[feature] = 0
        return feature_df
    
    def predict_risk(self, df):
        """Predict risk for companies"""
        features_df = self.prepare_features(df)
        features_imputed = self.imputer.transform(features_df)
        features_scaled = self.scaler.transform(features_imputed)
        predictions = self.model.predict(features_scaled)
        probabilities = self.model.predict_proba(features_scaled)
        return predictions, probabilities
    
    def add_predictions_to_df(self, df):
        """Add predictions as new columns"""
        predictions, probabilities = self.predict_risk(df)
        df_with_pred = df.copy()
        df_with_pred['ML_Prediction'] = predictions
        df_with_pred['ML_Prediction_Label'] = ['RISKY' if p == 1 else 'SAFE' for p in predictions]
        df_with_pred['ML_Risk_Probability'] = probabilities[:, 1]
        return df_with_pred
